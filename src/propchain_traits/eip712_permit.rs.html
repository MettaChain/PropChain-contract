5/// EIP-2612-style permit payload (Issue #995).
6///
7/// The struct is only a *payload description*: on its own it proves nothing.
8/// Authorization requires an ECDSA signature over [`Eip712Permit::message_hash`]
9/// that recovers to the owner's registered compressed public key, plus chain,
10/// nonce and deadline checks. See `verify_permit`.
11#[derive(Clone, Debug)]
12pub struct Eip712Permit {
13    pub chain_id: u64,
14    pub nonce: u64,
15    pub owner: AccountId,
16    pub spender: AccountId,
17    pub value: u128,
18    pub deadline: u64,
19}
20
21impl Eip712Permit {
22    /// Domain-separated digest binding *every* field of the permit
23    /// (chain id, nonce, owner, spender, value, deadline). A signature over
24    /// this hash cannot be replayed against a different spender, value or
25    /// chain because each is part of the signed message.
26    pub fn message_hash(&self) -> Hash {
27        crypto::hash_encoded(&(
28            b"PropChain:Eip712Permit:v1",
29            self.chain_id,
30            self.nonce,
31            self.owner,
32            self.spender,
33            self.value,
34            self.deadline,
35        ))
36    }
37
38    /// Fully verify a permit:
39    ///
40    /// 1. the permit targets the current chain (`current_chain_id`),
41    /// 2. its nonce matches the on-chain expectation (replay guard),
42    /// 3. it has not expired (`now <= deadline`),
43    /// 4. `signature` recovers to `owner_public_key` over the domain-separated
44    ///    [`Eip712Permit::message_hash`].
45    ///
46    /// The workspace account model stores explicit `[u8; 33]` public keys per
47    /// account (see governance `register_public_key`), so the caller resolves
48    /// `owner` to its registered key; recovery guarantees the signature was
49    /// produced by that key's secret. The spender/value are authenticated
50    /// transitively: they are part of the signed hash, so a signature valid
51    /// for one permit cannot authorize a different one.
52    pub fn verify_permit(
53        &self,
54        current_chain_id: u64,
55        expected_nonce: u64,
56        now: u64,
57        owner_public_key: &[u8; 33],
58        signature: &[u8; 65],
59    ) -> bool {
60        if self.chain_id != current_chain_id || self.nonce != expected_nonce {
61            return false;
62        }
63        if now > self.deadline {
64            return false;
65        }
66        let message_hash = <[u8; 32]>::from(self.message_hash());
67        match crypto::verify_ecdsa_signature(signature, &message_hash) {
68            Ok(recovered) => recovered == *owner_public_key,
69            Err(_) => false,
70        }
71    }
72}

80    /// Sign `hash` with the secret for `index` and return
81    /// (public_key, signature) in the shapes the contract expects.
82    fn sign(index: u8, hash: &[u8; 32]) -> ([u8; 33], [u8; 65]) {
83        let engine = Secp256k1::signing_only();
84        let mut secret = [0u8; 32];
85        secret[31] = index + 1; // deterministic distinct keys
86        let secret = SecretKey::from_slice(&secret).expect("valid secret");
87        let msg = Message::from_digest_slice(hash).expect("valid message");
88        let sig = engine.sign_ecdsa_recoverable(&msg, &secret);
89        let (recovery_id, data) = sig.serialize_compact();
90        let mut signature = [0u8; 65];
91        signature[..64].copy_from_slice(&data);
92        signature[64] = recovery_id.to_i32() as u8;
93        let public = secp256k1::PublicKey::from_secret_key(&engine, &secret);
94        (public.serialize(), signature)
95    }

97    fn sample_permit(owner: AccountId, spender: AccountId) -> Eip712Permit {

99            chain_id: 1,

108    #[test]
109    fn valid_signature_from_owner_passes() {
110        let owner = AccountId::from([0x11; 32]);
111        let spender = AccountId::from([0x22; 32]);
112        let permit = sample_permit(owner, spender);
113        let hash = <[u8; 32]>::from(permit.message_hash());
114        let (public_key, signature) = sign(1, &hash);
115
116        assert!(permit.verify_permit(1, 7, 4_999, &public_key, &signature));
117        // Boundary: at exactly the deadline the permit is still usable.
118        assert!(permit.verify_permit(1, 7, 5_000, &public_key, &signature));
119    }

121    #[test]
122    fn forged_signature_from_wrong_signer_fails() {
123        let owner = AccountId::from([0x11; 32]);
124        let spender = AccountId::from([0x22; 32]);
125        let permit = sample_permit(owner, spender);
126        let hash = <[u8; 32]>::from(permit.message_hash());
127
128        // Signed by someone who is not the owner.
129        let (_attacker_key, attacker_sig) = sign(2, &hash);
130        let (owner_key, _owner_sig) = sign(1, &hash);
131        assert!(!permit.verify_permit(1, 7, 4_999, &owner_key, &attacker_sig));
132    }

134    #[test]
135    fn signature_over_tampered_payload_fails() {
136        let owner = AccountId::from([0x11; 32]);
137        let spender = AccountId::from([0x22; 32]);
138        let permit = sample_permit(owner, spender);
139        let hash = <[u8; 32]>::from(permit.message_hash());
140        let (public_key, signature) = sign(1, &hash);
141
142        // Same key/signature but the permit now names a different spender or
143        // value: the payload hash no longer matches the signed one.
144        let mut tampered = sample_permit(owner, spender);
145        tampered.value = 999_999;
146        assert!(!tampered.verify_permit(1, 7, 4_999, &public_key, &signature));
147
148        let mut other_spender = sample_permit(owner, spender);
149        other_spender.spender = AccountId::from([0x33; 32]);
150        assert!(!other_spender.verify_permit(1, 7, 4_999, &public_key, &signature));
151    }

154    fn wrong_chain_nonce_or_deadline_fails() {

169    #[test]
170    fn message_hash_binds_all_fields() {
171        let base = sample_permit(AccountId::from([0x11; 32]), AccountId::from([0x22; 32]));
172        let base_hash = base.message_hash();
173
174        let mut changed = base.clone();
175        changed.value += 1;
176        assert_ne!(base_hash, changed.message_hash());
177
178        let mut changed = base.clone();
179        changed.deadline += 1;
180        assert_ne!(base_hash, changed.message_hash());
181
182        let mut changed = base.clone();
183        changed.spender = AccountId::from([0x44; 32]);
184        assert_ne!(base_hash, changed.message_hash());
