1//! Multicall types shared across the workspace.
2//!
3//! A `CallRequest` describes a single cross-contract call to be dispatched
4//! by the Multicall contract.  `CallResult` carries the outcome of each
5//! individual call so callers can inspect partial failures.
6//!
7//! Issue #737 substrate lives below in addition to the existing multicall
8//! types: `VerificationKind` and the pure-rust `aggregate_verifications`
9//! helper exist to support batching Identity/Compliance/Sanctions/Oracle
10//! checks through a single multicall. The actual on-chain dispatch
11//! remains in `contracts/multicall/src/lib.rs`; this file owns the
12//! types and the request-construction helper only.
13
14use ink::prelude::vec::Vec;
15
16/// A single call to be dispatched inside a multicall transaction.
17#[derive(Debug, Clone, PartialEq, Eq, scale::Encode, scale::Decode)]
18#[cfg_attr(
19    feature = "std",
20    derive(scale_info::TypeInfo, ink::storage::traits::StorageLayout)
21)]
22pub struct CallRequest {
23    /// Target contract address.
24    pub callee: ink::primitives::AccountId,
25    /// 4-byte selector followed by SCALE-encoded arguments.
26    pub selector_and_input: Vec<u8>,
27    /// Native token value to forward with the call (0 for most calls).
28    pub transferred_value: u128,
29    /// Gas limit for this individual call (0 = use remaining gas).
30    pub gas_limit: u64,
31    /// When `true` the entire multicall reverts if this call fails.
32    /// When `false` the failure is recorded and execution continues.
33    pub allow_revert: bool,
34}
35
36/// Outcome of a single dispatched call.
37#[derive(Debug, Clone, PartialEq, Eq, scale::Encode, scale::Decode)]
38#[cfg_attr(
39    feature = "std",
40    derive(scale_info::TypeInfo, ink::storage::traits::StorageLayout)
41)]
42pub struct CallResult {
43    /// Index of the originating `CallRequest` in the input slice.
44    pub index: u32,
45    /// Whether the call succeeded.
46    pub success: bool,
47    /// SCALE-encoded return data on success, or error bytes on failure.
48    pub return_data: Vec<u8>,
49}
50
51/// Errors returned by the Multicall contract.
52#[derive(Debug, Clone, PartialEq, Eq, scale::Encode, scale::Decode)]
53#[cfg_attr(feature = "std", derive(scale_info::TypeInfo))]
54pub enum MulticallError {
55    /// The calls vector was empty.
56    EmptyCalls,
57    /// The number of calls exceeds `MAX_MULTICALL_SIZE`.
58    TooManyCalls,
59    /// A call with `allow_revert = false` failed; index of the failing call
60    /// is embedded so the caller knows which one caused the revert.
61    CallReverted(u32),
62    /// The contract is paused.
63    Paused,
64    /// Caller is not the admin.
65    Unauthorized,
66}

68// ---------------------------------------------------------------------------
69// Issue #737 substrate: aggregation of onboarding verification checks.
70// ---------------------------------------------------------------------------
71
72/// Categories of verification checks that onboarding flows may want to
73/// batch through `Multicall::aggregate`.
74///
75/// `Identity`   — verify the actor is a real, registered identity.
76/// `Compliance` — verify the actor is compliant with jurisdictional rules.
77/// `Sanctions`  — verify the actor is not on any sanctions list.
78/// `Oracle`     — verify the property payload against an oracle feed.
79///
80/// These map 1:1 onto separate contracts in the workspace today. The
81/// intent of `aggregate_verifications` is to give any onboarding flow
82/// a single batched entry point so a caller composes one
83/// `Vec<CallRequest>` and hands it to `Multicall::aggregate`, instead of
84/// issuing N independent round-trip messages.
85#[derive(Debug, Clone, Copy, PartialEq, Eq, scale::Encode, scale::Decode)]
86#[cfg_attr(feature = "std", derive(scale_info::TypeInfo))]
87pub enum VerificationKind {
88    Identity,
89    Compliance,
90    Sanctions,
91    Oracle,
92}
93
94/// Length of a SCALE function selector preamble.
95pub const CALL_SELECTOR_LEN: usize = 4;
96
97/// Stub 4-byte SCALE selectors per `VerificationKind`. Replace with
98/// production selectors (e.g. `ink::selector_bytes!("verify")`) once
99/// the underlying verification contract messages are finalised; the
100/// stubs are deliberately distinct from each other so a future
101/// subcontract can replace one selector without breaking the others.
102pub fn verification_selector(kind: VerificationKind) -> [u8; CALL_SELECTOR_LEN] {
103    match kind {
104        VerificationKind::Identity => [0x01, 0x00, 0x00, 0x00],
105        VerificationKind::Compliance => [0x02, 0x00, 0x00, 0x00],
106        VerificationKind::Sanctions => [0x03, 0x00, 0x00, 0x00],
107        VerificationKind::Oracle => [0x04, 0x00, 0x00, 0x00],
108    }
109}
110
111/// Build one `CallRequest` for a verification kind targeting `callee`.
112///
113/// The 4-byte selector goes first; the caller's SCALE-encoded argument
114/// bytes follow verbatim as `selector_and_input`. The returned
115/// `CallRequest` is suitable for handing to `Multicall::aggregate`.
116pub fn build_verification_call(
117    callee: ink::primitives::AccountId,
118    kind: VerificationKind,
119    input: &[u8],
120) -> CallRequest {
121    let selector = verification_selector(kind);
122    let mut selector_and_input = Vec::with_capacity(CALL_SELECTOR_LEN + input.len());
123    selector_and_input.extend_from_slice(&selector);
124    selector_and_input.extend_from_slice(input);
125    CallRequest {
126        callee,
127        selector_and_input,
128        transferred_value: 0,
129        gas_limit: 0,
130        // Verification checks during onboarding should never abort the
131        // whole batch — the aggregator decides policy on partial failures.
132        allow_revert: true,
133    }
134}
135
136/// Build the slice of `CallRequest`s for an onboarding batch of
137/// verification checks. The result is intended to be passed directly
138/// into the `Multicall::aggregate` entry point (see
139/// `contracts/multicall/src/lib.rs`).
140///
141/// This function is a pure-rust constructor: it does NOT perform
142/// cross-contract calls itself. Caller dispatches `Vec<CallRequest>`
143/// to `Multicall::aggregate` at transaction time. The win over the
144/// previous four-message per-check pattern is that onboarding flows
145/// only have to compose one batch in memory instead of issuing N
146/// independent messages.
147pub fn aggregate_verifications(
148    callee: ink::primitives::AccountId,
149    requests: &[VerificationKind],
150    input: &[u8],
151) -> Vec<CallRequest> {
152    requests
153        .iter()
154        .map(|kind| build_verification_call(callee, *kind, input))
155        .collect()
156}

158#[cfg(test)]
159mod tests {
160    use super::*;
161    use ink::primitives::AccountId;
162
163    #[test]
164    fn selectors_are_distinct_across_kinds() {
165        let kinds = [
166            VerificationKind::Identity,
167            VerificationKind::Compliance,
168            VerificationKind::Sanctions,
169            VerificationKind::Oracle,
170        ];
171        for pair in kinds.windows(2) {
172            assert_ne!(
173                verification_selector(pair[0]),
174                verification_selector(pair[1]),
175                "selectors must be distinct across kinds so a multicall can disambiguate them"
176            );
177        }
178    }
179
180    #[test]
181    fn aggregate_verifications_yields_one_call_per_kind() {
182        let callee = AccountId::from([0xab; 32]);
183        let input = [0xa0u8, 0xa1, 0xa2];
184        let kinds = [
185            VerificationKind::Identity,
186            VerificationKind::Compliance,
187            VerificationKind::Sanctions,
188            VerificationKind::Oracle,
189        ];
190        let calls = aggregate_verifications(callee, &kinds, &input);
191        assert_eq!(calls.len(), kinds.len());
192        for call in &calls {
193            assert_eq!(call.callee, callee);
194            assert_eq!(call.transferred_value, 0);
195            assert_eq!(call.gas_limit, 0);
196            assert!(call.allow_revert);
197        }
198    }
199
200    #[test]
201    fn aggregate_verifications_appends_input_after_selector_bytes() {
202        let callee = AccountId::from([0xab; 32]);
203        let input = [0xf0u8, 0xf1, 0xf2, 0xf3];
204        let kinds = [VerificationKind::Identity];
205        let calls = aggregate_verifications(callee, &kinds, &input);
206        assert_eq!(calls.len(), 1);
207        let call = &calls[0];
208        assert_eq!(
209            call.selector_and_input.len(),
210            CALL_SELECTOR_LEN + input.len()
211        );
212        // Identity selector bytes are first.
213        assert_eq!(
214            &call.selector_and_input[..CALL_SELECTOR_LEN],
215            &[0x01u8, 0x00, 0x00, 0x00][..]
216        );
217        // Caller's SCALE-encoded args follow verbatim.
218        assert_eq!(&call.selector_and_input[CALL_SELECTOR_LEN..], &input[..]);
219    }
220
221    #[test]
222    fn empty_kinds_slice_yields_no_calls() {
223        let callee = AccountId::from([0xab; 32]);
224        let calls = aggregate_verifications(callee, &[], &[1, 2, 3]);
225        assert!(calls.is_empty());
226    }
