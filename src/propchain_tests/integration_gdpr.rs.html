1/// # Integration Tests: GDPR Consent Management (Issue #1005)
2///
3/// These tests verify the end-to-end consent lifecycle of the
4/// `propchain-gdpr` contract:
5///   grant -> check -> withdraw -> re-check
6/// plus the data access request flow and input validation guards.
7///
8/// Acceptance criteria tested:
9///   check Admin grants consent for a data subject with a positive duration
10///   check check_consent reports true while a consent is granted and unexpired
11///   check Data subject can withdraw their own consent; check_consent flips to false
12///   check Admin can withdraw on behalf of a subject; third parties are rejected
13///   check Data subject requests data access; admin fulfills it
14///   check Non-admin cannot fulfill a data access request
15///   check get_data_access_request reflects fulfilled status afterwards
16///   check Zero-duration consent grants are rejected with InvalidDuration
17#[cfg(test)]
18#[allow(clippy::module_inception)]
19mod integration_gdpr {
20    // GDPR consent contract
21    use ink::env::{test, DefaultEnvironment};
22    use propchain_gdpr::gdpr_consent::{
23        ConsentStatus, Error as GdprError, GdprConsent, ProcessingPurpose,
24    };
25
26    /// Duration used for "valid" grants (1 year in milliseconds).
27    const ONE_YEAR_MS: u64 = 365 * 24 * 60 * 60 * 1000;
28
29    /// Deploy the contract with `accounts.alice` as admin.
30    fn setup() -> GdprConsent {
31        let accounts = test::default_accounts::<DefaultEnvironment>();
32        test::set_caller::<DefaultEnvironment>(accounts.alice);
33        GdprConsent::new()
34    }
35
36    /// Scenario 1 - Full consent lifecycle initiated by the admin
37    /// 1. Admin grants KYC consent for bob
38    /// 2. check_consent(bob, KYC) is true
39    /// 3. Bob withdraws his own consent
40    /// 4. check_consent(bob, KYC) is false again
41    #[ink::test]
42    fn test_grant_check_withdraw_consent_lifecycle() {
43        let accounts = test::default_accounts::<DefaultEnvironment>();
44        let mut gdpr = setup();
45
46        let consent_id = gdpr
47            .grant_consent(accounts.bob, ProcessingPurpose::KYC, ONE_YEAR_MS)
48            .expect("Admin should be able to grant consent");
49
50        assert!(
51            gdpr.check_consent(accounts.bob, ProcessingPurpose::KYC),
52            "Consent must be active right after granting"
53        );
54        assert!(
55            !gdpr.check_consent(accounts.bob, ProcessingPurpose::Marketing),
56            "Consent must be purpose-scoped: other purposes stay false"
57        );
58
59        // Subject withdraws their own consent
60        test::set_caller::<DefaultEnvironment>(accounts.bob);
61        gdpr.withdraw_consent(consent_id)
62            .expect("Data subject must be able to withdraw own consent");
63
64        assert!(
65            !gdpr.check_consent(accounts.bob, ProcessingPurpose::KYC),
66            "Withdrawn consent must fail check_consent"
67        );
68
69        let record = gdpr.get_consent(consent_id).expect("Record should persist");
70        assert_eq!(record.status, ConsentStatus::Withdrawn);
71        assert!(
72            record.withdrawn_at.is_some(),
73            "withdrawn_at must be recorded"
74        );
75    }

77    /// Scenario 2 - Withdrawal authorization matrix:
78    /// subject and admin may withdraw, unrelated parties may not.
79    #[ink::test]
80    fn test_withdraw_authorization_subject_admin_only() {
81        let accounts = test::default_accounts::<DefaultEnvironment>();
82        let mut gdpr = setup();
83
84        let id_bob = gdpr
85            .grant_consent(accounts.bob, ProcessingPurpose::TaxReporting, ONE_YEAR_MS)
86            .expect("Grant for bob should succeed");
87        let id_charlie = gdpr
88            .grant_consent(
89                accounts.charlie,
90                ProcessingPurpose::RiskAssessment,
91                ONE_YEAR_MS,
92            )
93            .expect("Grant for charlie should succeed");
94
95        // Unrelated party (charlie) cannot withdraw bob's consent
96        test::set_caller::<DefaultEnvironment>(accounts.charlie);
97        assert_eq!(
98            gdpr.withdraw_consent(id_bob),
99            Err(GdprError::NotAuthorized),
100            "Third party must not withdraw someone else's consent"
101        );
102
103        // Subject (bob) CAN withdraw
104        test::set_caller::<DefaultEnvironment>(accounts.bob);
105        gdpr.withdraw_consent(id_bob)
106            .expect("Subject must be able to withdraw own consent");
107
108        // Admin (alice) CAN withdraw on behalf of charlie
109        test::set_caller::<DefaultEnvironment>(accounts.alice);
110        gdpr.withdraw_consent(id_charlie)
111            .expect("Admin must be able to withdraw consent on behalf of a subject");
112
113        assert!(!gdpr.check_consent(accounts.bob, ProcessingPurpose::TaxReporting));
114        assert!(!gdpr.check_consent(accounts.charlie, ProcessingPurpose::RiskAssessment));
115    }

117    /// Scenario 3 - Data access request flow:
118    /// subject requests -> non-admin fulfill rejected -> admin fulfills.
119    #[ink::test]
120    fn test_data_access_request_fulfillment_flow() {
121        let accounts = test::default_accounts::<DefaultEnvironment>();
122        let mut gdpr = setup();
123
124        // Bob submits a data access request
125        test::set_caller::<DefaultEnvironment>(accounts.bob);
126        let request_id = gdpr
127            .request_data_access()
128            .expect("Subject should request access");
129
130        let pending = gdpr
131            .get_data_access_request(request_id)
132            .expect("Request should exist");
133        assert_eq!(pending.data_subject, accounts.bob);
134        assert!(!pending.fulfilled, "Request must start unfulfilled");
135
136        // Non-admin fulfillment attempt is rejected
137        test::set_caller::<DefaultEnvironment>(accounts.bob);
138        assert_eq!(
139            gdpr.fulfill_data_access(request_id),
140            Err(GdprError::NotAuthorized),
141            "Only the admin may fulfill data access requests"
142        );
143        assert!(
144            !gdpr.get_data_access_request(request_id).unwrap().fulfilled,
145            "Rejected fulfillment must not mutate state"
146        );
147
148        // Admin fulfills the request
149        test::set_caller::<DefaultEnvironment>(accounts.alice);
150        gdpr.fulfill_data_access(request_id)
151            .expect("Admin should fulfill the data access request");
152
153        let fulfilled = gdpr
154            .get_data_access_request(request_id)
155            .expect("Request should exist");
156        assert!(fulfilled.fulfilled, "Request must be marked fulfilled");
157        assert!(
158            fulfilled.fulfilled_at.is_some(),
159            "fulfilled_at must be recorded"
160        );
161
162        // The subject sees the fulfilled request in their history
163        let history = gdpr.get_subject_requests(accounts.bob);
164        assert_eq!(history.len(), 1);
165        assert!(history[0].fulfilled);
166    }

168    /// Scenario 4 - Input validation: zero-duration grants rejected.
169    #[ink::test]
170    fn test_zero_duration_grant_rejected() {
171        let accounts = test::default_accounts::<DefaultEnvironment>();
172        let mut gdpr = setup();
173
174        assert_eq!(
175            gdpr.grant_consent(accounts.bob, ProcessingPurpose::KYC, 0),
176            Err(GdprError::InvalidDuration),
177            "Zero-duration consent must be rejected"
178        );
179
180        // Nothing was stored for the subject
181        assert!(!gdpr.check_consent(accounts.bob, ProcessingPurpose::KYC));
182        assert!(gdpr.get_subject_consents(accounts.bob).is_empty());
183    }

185    /// Scenario 5 - Multiple purposes tracked per subject independently;
186    /// withdrawing one purpose leaves others intact.
187    #[ink::test]
188    fn test_multi_purpose_consents_are_independent() {
189        let accounts = test::default_accounts::<DefaultEnvironment>();
190        let mut gdpr = setup();
191
192        gdpr.grant_consent(accounts.bob, ProcessingPurpose::KYC, ONE_YEAR_MS)
193            .expect("KYC grant should succeed");
194        let marketing_id = gdpr
195            .grant_consent(accounts.bob, ProcessingPurpose::Marketing, ONE_YEAR_MS)
196            .expect("Marketing grant should succeed");
197
198        let records = gdpr.get_subject_consents(accounts.bob);
199        assert_eq!(records.len(), 2, "Both consents should be listed");
200        assert!(records.iter().all(|r| r.status == ConsentStatus::Granted));
201
202        // Withdrawing marketing keeps KYC active
203        test::set_caller::<DefaultEnvironment>(accounts.bob);
204        gdpr.withdraw_consent(marketing_id)
205            .expect("Withdraw should succeed");
206
207        assert!(
208            gdpr.check_consent(accounts.bob, ProcessingPurpose::KYC),
209            "KYC consent must remain granted"
210        );
211        assert!(
212            !gdpr.check_consent(accounts.bob, ProcessingPurpose::Marketing),
213            "Marketing consent must be withdrawn"
214        );
215    }
