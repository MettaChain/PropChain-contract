1/// # Integration Tests: Governance Multisig Proposal Lifecycle (Issue #1002)
2///
3/// These tests verify the end-to-end governance flow:
4///   signer setup -> create_proposal -> votes to threshold -> timelock ->
5///   execute_proposal
6///
7/// Because ink! unit tests run inside a single contract environment, we test
8/// the contract directly rather than through cross-contract calls. This
9/// mirrors the actual interaction semantics.
10///
11/// Acceptance criteria tested:
12///   check Signer setup via constructor (admin = deployer)
13///   check Proposal creation by a signer
14///   check Votes by signers until the approval threshold is met
15///   check Execution is rejected before threshold and while timelock is active
16///   check Timelock elapses after advancing blocks; execution then succeeds
17///   check Non-signers cannot propose or vote
18///   check Double voting is rejected
19///   check Certain majority-against proposals are rejected and not executable
20#[cfg(test)]
21#[allow(clippy::module_inception)]
22mod integration_governance {
23    use governance::governance::{
24        Error as GovernanceError, Governance, GovernanceAction, ProposalStatus,
25    };
26    use ink::env::{test, DefaultEnvironment};
27    use ink::primitives::Hash;
28
29    const TIMELOCK_BLOCKS: u64 = 10;
30
31    fn description_hash(byte: u8) -> Hash {
32        Hash::from([byte; 32])
33    }
34
35    /// Two signers (alice, bob), threshold 2. Alice deploys and becomes admin.
36    fn setup() -> Governance {
37        let accounts = test::default_accounts::<DefaultEnvironment>();
38        test::set_caller::<DefaultEnvironment>(accounts.alice);
39        Governance::new(vec![accounts.alice, accounts.bob], 2, TIMELOCK_BLOCKS)
40    }
41
42    /// Three signers (alice, bob, charlie), threshold 2.
43    fn setup_three_signers() -> Governance {
44        let accounts = test::default_accounts::<DefaultEnvironment>();
45        test::set_caller::<DefaultEnvironment>(accounts.alice);
46        Governance::new(
47            vec![accounts.alice, accounts.bob, accounts.charlie],
48            2,
49            TIMELOCK_BLOCKS,
50        )
51    }
52
53    fn create_proposal(governance: &mut Governance) -> u64 {
54        let accounts = test::default_accounts::<DefaultEnvironment>();
55        test::set_caller::<DefaultEnvironment>(accounts.alice);
56        governance
57            .create_proposal(
58                description_hash(0xA1),
59                GovernanceAction::ChangeThreshold,
60                None,
61            )
62            .expect("Signer should be able to create a proposal")
63    }
64
65    /// Scenario 1 - Happy path: propose → vote to threshold → timelock → execute
66    #[ink::test]
67    fn test_proposal_vote_threshold_timelock_then_execute_succeeds() {
68        let accounts = test::default_accounts::<DefaultEnvironment>();
69        let mut governance = setup();
70
71        // Constructor sanity: admin is deployer, signers + threshold stored.
72        assert_eq!(governance.get_admin(), accounts.alice);
73        assert_eq!(governance.get_signers(), vec![accounts.alice, accounts.bob]);
74        assert_eq!(governance.get_threshold(), 2);
75
76        // Step 1: signer creates a proposal
77        let proposal_id = create_proposal(&mut governance);
78        let proposal = governance
79            .get_proposal(proposal_id)
80            .expect("Proposal should exist after creation");
81        assert_eq!(proposal.status, ProposalStatus::Active);
82        assert_eq!(proposal.proposer, accounts.alice);
83
84        // Step 2: execution before threshold must be rejected (still Active)
85        test::set_caller::<DefaultEnvironment>(accounts.alice);
86        assert_eq!(
87            governance.execute_proposal(proposal_id),
88            Err(GovernanceError::ProposalClosed),
89            "Execution must be rejected before approval threshold"
90        );
91
92        // Step 3: first approval vote - below threshold, still Active
93        test::set_caller::<DefaultEnvironment>(accounts.alice);
94        governance
95            .vote(proposal_id, true)
96            .expect("Signer vote should succeed");
97
98        // Step 4: second approval vote reaches threshold → Approved + timelock
99        test::set_caller::<DefaultEnvironment>(accounts.bob);
100        governance
101            .vote(proposal_id, true)
102            .expect("Second signer vote should succeed");
103        let proposal = governance.get_proposal(proposal_id).unwrap();
104        assert_eq!(proposal.status, ProposalStatus::Approved);
105        assert_eq!(
106            proposal.timelock_until,
107            proposal.created_at + TIMELOCK_BLOCKS,
108            "Timelock must start when threshold is reached"
109        );
110
111        // Step 5: execution during timelock must be rejected
112        test::set_caller::<DefaultEnvironment>(accounts.alice);
113        assert_eq!(
114            governance.execute_proposal(proposal_id),
115            Err(GovernanceError::TimelockActive),
116            "Execution must be blocked while the timelock is active"
117        );
118
119        // Step 6: advance past the timelock window, execution succeeds
120        let current_block = ink::env::block_number::<DefaultEnvironment>() as u64;
121        test::set_block_number::<DefaultEnvironment>(
122            (current_block.max(proposal.timelock_until) + 1) as u32,
123        );
124        governance
125            .execute_proposal(proposal_id)
126            .expect("Execution should succeed once timelock has elapsed");
127
128        let executed = governance.get_proposal(proposal_id).unwrap();
129        assert_eq!(executed.status, ProposalStatus::Executed);
130        assert!(executed.executed_at > 0);
131        assert_eq!(governance.get_active_proposal_count(), 0);
132    }

134    /// Scenario 2 - Non-signers cannot create proposals nor vote
135    #[ink::test]
136    fn test_non_signer_cannot_propose_or_vote() {
137        let accounts = test::default_accounts::<DefaultEnvironment>();
138        let mut governance = setup();
139
140        test::set_caller::<DefaultEnvironment>(accounts.charlie);
141        assert_eq!(
142            governance.create_proposal(
143                description_hash(0xB2),
144                GovernanceAction::AddSigner,
145                Some(accounts.charlie),
146            ),
147            Err(GovernanceError::NotASigner),
148            "Non-signer must not create proposals"
149        );
150
151        let proposal_id = create_proposal(&mut governance);
152
153        test::set_caller::<DefaultEnvironment>(accounts.charlie);
154        assert_eq!(
155            governance.vote(proposal_id, true),
156            Err(GovernanceError::NotASigner),
157            "Non-signer must not vote"
158        );
159
160        // The invalid vote must not have been recorded.
161        let proposal = governance.get_proposal(proposal_id).unwrap();
162        assert_eq!(proposal.votes_for, 0);
163        assert_eq!(proposal.votes_against, 0);
164    }

166    /// Scenario 3 - Double voting by the same signer is rejected
167    #[ink::test]
168    fn test_double_vote_is_rejected() {
169        let accounts = test::default_accounts::<DefaultEnvironment>();
170        let mut governance = setup();
171
172        let proposal_id = create_proposal(&mut governance);
173
174        test::set_caller::<DefaultEnvironment>(accounts.alice);
175        governance
176            .vote(proposal_id, true)
177            .expect("First vote should succeed");
178
179        assert_eq!(
180            governance.vote(proposal_id, true),
181            Err(GovernanceError::AlreadyVoted),
182            "Duplicate vote by the same signer must be rejected"
183        );
184        assert_eq!(
185            governance.vote(proposal_id, false),
186            Err(GovernanceError::AlreadyVoted),
187            "Vote flip via duplicate ballot must also be rejected"
188        );
189
190        let proposal = governance.get_proposal(proposal_id).unwrap();
191        assert_eq!(proposal.votes_for, 1, "Only one vote must be counted");
192        assert_eq!(proposal.votes_against, 0);
193    }
194
195    /// Scenario 4 - Majority-against proposal becomes unrecoverable and
196    /// cannot be executed afterwards
197    #[ink::test]
198    fn test_majority_against_rejects_proposal_and_blocks_execution() {
199        let accounts = test::default_accounts::<DefaultEnvironment>();
200        let mut governance = setup_three_signers();
201
202        let proposal_id = create_proposal(&mut governance);
203
204        // Two rejections out of three signers make approval mathematically
205        // impossible: remaining votes (1) + votes_for (0) < threshold (2).
206        test::set_caller::<DefaultEnvironment>(accounts.alice);
207        governance
208            .vote(proposal_id, false)
209            .expect("Alice reject works");
210
211        // Not yet decided: approval is still theoretically possible.
212        assert_eq!(
213            governance.get_proposal(proposal_id).unwrap().status,
214            ProposalStatus::Active
215        );
216
217        test::set_caller::<DefaultEnvironment>(accounts.bob);
218        governance
219            .vote(proposal_id, false)
220            .expect("Bob reject works");
221
222        let rejected = governance.get_proposal(proposal_id).unwrap();
223        assert_eq!(rejected.status, ProposalStatus::Rejected);
224        assert_eq!(rejected.votes_against, 2);
225
226        test::set_caller::<DefaultEnvironment>(accounts.alice);
227        assert_eq!(
228            governance.execute_proposal(proposal_id),
229            Err(GovernanceError::ProposalClosed),
230            "A rejected proposal must not be executable"
231        );
232    }
