222        // ── Consent Management ──────────────────────────────────────────────

224        /// Grants consent for a data subject and processing purpose.
225        ///
226        /// Only the data subject themself may grant consent for their own
227        /// data; the contract admin may additionally record consent on behalf
228        /// of a subject (mirroring [`withdraw_consent`]'s authorization rule).
229        /// Any other caller is rejected with [`Error::NotAuthorized`] so
230        /// third parties cannot fabricate consent records in someone else's
231        /// name.
232        ///
233        /// The record expires after `duration_ms` milliseconds from the
234        /// current block timestamp and its id is appended to the subject's
235        /// consent list (see [`Self::get_subject_consents`]).
236        ///
237        /// # Errors
238        /// - [`Error::NotAuthorized`] when the caller is neither the subject
239        ///   nor the admin.
240        /// - [`Error::InvalidDuration`] when `duration_ms` is zero.
241        #[ink(message)]
242        pub fn grant_consent(
243            &mut self,
244            data_subject: AccountId,
245            purpose: ProcessingPurpose,
246            duration_ms: u64,
247        ) -> Result<u64> {
248            let caller = self.env().caller();
249            if caller != data_subject && caller != self.admin {
250                return Err(Error::NotAuthorized);
251            }
252
253            if duration_ms == 0 {
254                return Err(Error::InvalidDuration);
255            }
256
257            let consent_id = self.next_consent_id;
258            self.next_consent_id = consent_id.checked_add(1).ok_or(Error::InvalidDuration)?;
259
260            let now = self.now();
261            let record = ConsentRecord {
262                consent_id,
263                data_subject,
264                processor: self.admin,
265                purpose: purpose.clone(),
266                status: ConsentStatus::Granted,
267                granted_at: now,
268                expires_at: now.checked_add(duration_ms).ok_or(Error::InvalidDuration)?,
269                withdrawn_at: None,
270            };
271            self.consent_records.insert(consent_id, &record);
272
273            let mut consents = self.subject_consents.get(data_subject).unwrap_or_default();
274            consents.push(consent_id);
275            self.subject_consents.insert(data_subject, &consents);
276
277            self.env().emit_event(ConsentGranted {
278                data_subject,
279                consent_id,
280                purpose,
281                timestamp: now,
282            });
283            Ok(consent_id)
284        }
285
286        /// Revokes a previously granted consent.
287        ///
288        /// Callable by the data subject themself or by the contract admin
289        /// (e.g. regulator-triggered erasure). Withdrawal takes effect
290        /// immediately: [`Self::check_consent`] returns `false` for the
291        /// record afterwards, and the withdrawal timestamp is recorded for
292        /// audit purposes. A consent can only be withdrawn once.
293        ///
294        /// # Errors
295        /// - [`Error::ConsentNotFound`] when `consent_id` is unknown or the
296        ///   record is no longer in the `Granted` state.
297        /// - [`Error::NotAuthorized`] when the caller is neither the subject
298        ///   nor the admin.
299        #[ink(message)]
300        pub fn withdraw_consent(&mut self, consent_id: u64) -> Result<()> {
301            let caller = self.env().caller();
302            let mut record = self
303                .consent_records
304                .get(consent_id)
305                .ok_or(Error::ConsentNotFound)?;
306
307            if record.data_subject != caller && caller != self.admin {
308                return Err(Error::NotAuthorized);
309            }
310            if record.status != ConsentStatus::Granted {
311                return Err(Error::ConsentNotFound);
312            }
313
314            let now = self.now();
315            record.status = ConsentStatus::Withdrawn;
316            record.withdrawn_at = Some(now);
317            self.consent_records.insert(consent_id, &record);
318
319            self.env().emit_event(ConsentWithdrawn {
320                data_subject: record.data_subject,
321                consent_id,
322                purpose: record.purpose.clone(),
323                timestamp: now,
324            });
325            Ok(())
326        }
327
328        /// Returns the full consent record for `consent_id`, or `None` if
329        /// unknown.
330        ///
331        /// The record exposes subject, processor, purpose, status and both
332        /// grant/expiry timestamps so integrators can render audit trails.
333        /// Note that a record with status [`ConsentStatus::Granted`] may
334        /// still be past its `expires_at`; use [`Self::check_consent`] for
335        /// the effective validity answer.
336        #[ink(message)]
337        pub fn get_consent(&self, consent_id: u64) -> Option<ConsentRecord> {
338            self.consent_records.get(consent_id)
339        }
340
341        /// Lists every consent record ever granted for `data_subject`,
342        /// regardless of current status (granted, withdrawn, expired).
343        ///
344        /// Read-only; order follows grant order. Returns an empty vector for
345        /// unknown subjects.
346        #[ink(message)]
347        pub fn get_subject_consents(&self, data_subject: AccountId) -> Vec<ConsentRecord> {
348            match self.subject_consents.get(data_subject) {
349                Some(ids) => {
350                    let mut records = Vec::new();
351                    for id in ids {
352                        if let Some(r) = self.consent_records.get(id) {
353                            records.push(r);
354                        }
355                    }
356                    records
357                }
358                None => Vec::new(),
359            }
360        }
361
362        /// Effective validity check: `true` only when the subject holds a
363        /// consent for `purpose` that is currently `Granted` **and** not yet
364        /// past its expiry timestamp.
365        ///
366        /// This is the predicate processors must consult before any
367        /// purpose-scoped data operation. Withdrawn and expired consents
368        /// both yield `false`.
369        #[ink(message)]
370        pub fn check_consent(&self, data_subject: AccountId, purpose: ProcessingPurpose) -> bool {
371            match self.subject_consents.get(data_subject) {
372                Some(ids) => {
373                    for id in ids {
374                        if let Some(record) = self.consent_records.get(id) {
375                            if record.purpose == purpose
376                                && record.status == ConsentStatus::Granted
377                                && record.expires_at > self.now()
378                            {
379                                return true;
380                            }
381                        }
382                    }
383                    false
384                }
385                None => false,
386            }
387        }
388
389        // ── Expiry Management (admin) ───────────────────────────────────────
390
391        /// Admin-only lifecycle transition: marks an already-stale consent
392        /// as `Expired`.
393        ///
394        /// The record must be in the `Granted` state **and** its expiry must
395        /// have passed (`expires_at <= now`); consents inside their validity
396        /// window cannot be force-expired. Expiring twice is rejected.
397        ///
398        /// # Errors
399        /// - [`Error::NotAuthorized`] when the caller is not the admin.
400        /// - [`Error::ConsentNotFound`] when the id is unknown, the consent
401        ///   is not yet stale, or it is no longer `Granted`.
402        #[ink(message)]
403        pub fn expire_consent(&mut self, consent_id: u64) -> Result<()> {
404            self.ensure_admin()?;
405            let mut record = self
406                .consent_records
407                .get(consent_id)
408                .ok_or(Error::ConsentNotFound)?;
409            let now = self.now();
410            if record.status != ConsentStatus::Granted || record.expires_at > now {
411                return Err(Error::ConsentNotFound);
412            }
413            record.status = ConsentStatus::Expired;
414            self.consent_records.insert(consent_id, &record);
415            self.env().emit_event(ConsentExpired {
416                data_subject: record.data_subject,
417                consent_id,
418                timestamp: now,
419            });
420            Ok(())
421        }

423        // ── Retention Policies ──────────────────────────────────────────────
424
425        /// Admin-only: stores the retention policy for a processing purpose.
426        ///
427        /// `retention_days` bounds how long data collected under `purpose`
428        /// may be kept, and `auto_delete` declares whether expiry triggers
429        /// automatic erasure. Re-setting the same purpose overwrites the
430        /// previous policy (the update is emitted as
431        /// `RetentionPolicyUpdated`).
432        ///
433        /// # Errors
434        /// - [`Error::NotAuthorized`] when the caller is not the admin.
435        #[ink(message)]
436        pub fn set_retention_policy(
437            &mut self,
438            purpose: ProcessingPurpose,
439            retention_days: u64,
440            auto_delete: bool,
441        ) -> Result<()> {
442            self.ensure_admin()?;
443            let key = Self::purpose_key(&purpose);
444            let policy = DataRetentionPolicy {
445                purpose: purpose.clone(),
446                retention_days,
447                auto_delete,
448            };
449            self.retention_policies.insert(key, &policy);
450            self.env().emit_event(RetentionPolicyUpdated {
451                purpose,
452                retention_days,
453                timestamp: self.now(),
454            });
455            Ok(())
456        }

458        /// Returns the retention policy configured for `purpose`, or `None`
459        /// when no policy has been set.
460        ///
461        /// Integrators should treat `None` as "no data may be retained
462        /// beyond the strict operational minimum" until a policy is
463        /// published.
464        #[ink(message)]
465        pub fn get_retention_policy(
466            &self,
467            purpose: ProcessingPurpose,
468        ) -> Option<DataRetentionPolicy> {
469            self.retention_policies.get(Self::purpose_key(&purpose))
470        }

472        // ── Data Access Requests ────────────────────────────────────────────
473
474        /// Data-subject access request (GDPR art. 15): the caller files a
475        /// request for disclosure of all personal data held about them.
476        ///
477        /// The caller is always the data subject; no admin involvement is
478        /// needed to file. The returned id can be tracked via
479        /// [`Self::get_data_access_request`] until the admin fulfils it.
480        #[ink(message)]
481        pub fn request_data_access(&mut self) -> Result<u64> {
482            let caller = self.env().caller();
483            let request_id = self.next_request_id;
484            self.next_request_id = request_id.checked_add(1).ok_or(Error::InvalidDuration)?;
485
486            let request = DataAccessRequest {
487                request_id,
488                data_subject: caller,
489                requested_at: self.now(),
490                fulfilled: false,
491                fulfilled_at: None,
492            };
493            self.data_access_requests.insert(request_id, &request);
494
495            let mut requests = self.subject_requests.get(caller).unwrap_or_default();
496            requests.push(request_id);
497            self.subject_requests.insert(caller, &requests);
498
499            self.env().emit_event(DataAccessRequested {
500                data_subject: caller,
501                request_id,
502                timestamp: self.now(),
503            });
504            Ok(request_id)
505        }
506
507        /// Admin-only: marks a data-access request as fulfilled and stamps
508        /// the fulfilment timestamp.
509        ///
510        /// # Errors
511        /// - [`Error::NotAuthorized`] when the caller is not the admin.
512        /// - [`Error::DataRequestNotFound`] when `request_id` is unknown.
513        #[ink(message)]
514        pub fn fulfill_data_access(&mut self, request_id: u64) -> Result<()> {
515            self.ensure_admin()?;
516            let mut request = self
517                .data_access_requests
518                .get(request_id)
519                .ok_or(Error::DataRequestNotFound)?;
520            request.fulfilled = true;
521            request.fulfilled_at = Some(self.now());
522            self.data_access_requests.insert(request_id, &request);
523            self.env().emit_event(DataAccessFulfilled {
524                data_subject: request.data_subject,
525                request_id,
526                timestamp: self.now(),
527            });
528            Ok(())
529        }
530
531        /// Returns the data-access request for `request_id`, or `None` if
532        /// unknown. Includes the fulfilment flag and timestamp.
533        #[ink(message)]
534        pub fn get_data_access_request(&self, request_id: u64) -> Option<DataAccessRequest> {
535            self.data_access_requests.get(request_id)
536        }
537
538        /// Lists every access request filed by `data_subject`, in filing
539        /// order. Empty vector for subjects that never filed.
540        #[ink(message)]
541        pub fn get_subject_requests(&self, data_subject: AccountId) -> Vec<DataAccessRequest> {
542            match self.subject_requests.get(data_subject) {
543                Some(ids) => {
544                    let mut requests = Vec::new();
545                    for id in ids {
546                        if let Some(r) = self.data_access_requests.get(id) {
547                            requests.push(r);
548                        }
549                    }
550                    requests
551                }
552                None => Vec::new(),
553            }
554        }

560            self.admin
