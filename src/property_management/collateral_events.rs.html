1//! Closes #807: move historical collateral assignments off-chain via events
2//! instead of an unbounded `collateral_history` Vec. Starter event struct +
3//! builder; wiring `env.emit_event(...)` into the assignment path and
4//! dropping the Vec field are follow-ups.
5
6pub struct CollateralAssignmentEvent {
7    pub property_id: u64,
8    pub collateral_id: u64,
9    pub assigned_at: u64,
10    pub active: bool,
11}
12
13/// Builds the event payload for a collateral assignment change, replacing
14/// an on-chain `Vec` push with something an off-chain indexer can consume.
15pub fn build_assignment_event(
16    property_id: u64,
17    collateral_id: u64,
18    assigned_at: u64,
19    active: bool,
20) -> CollateralAssignmentEvent {
21    CollateralAssignmentEvent { property_id, collateral_id, assigned_at, active }
22}
23
24#[cfg(test)]
25mod tests {
26    use super::*;
27
28    #[test]
29    fn builds_an_active_assignment_event() {
30        let event = build_assignment_event(1, 2, 1_000, true);
31        assert_eq!(event.property_id, 1);
32        assert_eq!(event.collateral_id, 2);
33        assert!(event.active);
34    }
35
36    #[test]
37    fn builds_a_deactivation_event() {
38        let event = build_assignment_event(1, 2, 2_000, false);
39        assert!(!event.active);
40    }
41
42    // ── Issue #1037: Additional collateral-event coverage ────────────────────
43
44    /// The `assigned_at` timestamp is stored and readable.
45    #[test]
46    fn event_preserves_assigned_at_timestamp() {
47        let ts = 9_999_999u64;
48        let event = build_assignment_event(5, 10, ts, true);
49        assert_eq!(event.assigned_at, ts);
50    }
51
52    /// All four fields survive the round-trip through `build_assignment_event`.
53    #[test]
54    fn event_fields_match_inputs_exactly() {
55        let event = build_assignment_event(42, 77, 123_456, false);
56        assert_eq!(event.property_id, 42);
57        assert_eq!(event.collateral_id, 77);
58        assert_eq!(event.assigned_at, 123_456);
59        assert!(!event.active);
60    }
61
62    /// Building multiple events for the same property with different collateral
63    /// ids produces independent records (no accidental aliasing).
64    #[test]
65    fn multiple_events_for_same_property_are_independent() {
66        let e1 = build_assignment_event(1, 10, 100, true);
67        let e2 = build_assignment_event(1, 20, 200, false);
68
69        assert_eq!(e1.collateral_id, 10);
70        assert_eq!(e2.collateral_id, 20);
71        assert_ne!(e1.assigned_at, e2.assigned_at);
72        assert!(e1.active);
73        assert!(!e2.active);
74    }
75
76    /// Transitioning the same (property, collateral) pair from active to
77    /// inactive is expressed by two separate event payloads; neither mutates
78    /// the other.
79    #[test]
80    fn activation_then_deactivation_events_are_independent() {
81        let activated = build_assignment_event(3, 7, 500, true);
82        let deactivated = build_assignment_event(3, 7, 600, false);
83
84        assert!(activated.active);
85        assert!(!deactivated.active);
86        // The timestamps differ — each event is its own snapshot.
87        assert_ne!(activated.assigned_at, deactivated.assigned_at);
88    }
89
90    /// `assigned_at = 0` is a valid sentinel value (e.g. genesis or
91    /// unset); the builder must not reject it.
92    #[test]
93    fn zero_assigned_at_is_accepted() {
94        let event = build_assignment_event(0, 0, 0, false);
95        assert_eq!(event.assigned_at, 0);
96        assert_eq!(event.property_id, 0);
97        assert_eq!(event.collateral_id, 0);
98    }
99}
