7/// Dynamic premium calculation with comprehensive risk factors.
8///
9/// Steps:
10///  1. Base rate (actuarial model or coverage-type default)
11///  2. Risk multiplier (weighted location / construction / age / claims-history scores)
12///  3. Coverage-type multiplier
13///  4. Pool-utilization multiplier
14///  5. Time-based multiplier
15///  6. Discount multiplier (multi-policy, claim-free, safety, loyalty)
16///  7. **Claim-frequency multiplier** (rolling-window surcharge — new)
17///
18/// The claim-frequency step uses `modifiers.recent_claims_count`, which the
19/// caller populates with the number of approved claims filed against the
20/// property within the rolling observation window (default: 12 months).
21/// Defaults match the table in `DYNAMIC_PREMIUM_CALCULATION.md`.
22pub fn calculate_dynamic_premium(
23    risk_assessment: &RiskAssessment,
24    coverage_amount: u128,
25    coverage_type: &CoverageType,
26    pool: &RiskPool,
27    actuarial_model: Option<&ActuarialModel>,
28    modifiers: &PremiumModifiers,
29    policy_duration_seconds: u64,
30) -> PremiumCalculation {
31    // Step 1: Calculate base rate from actuarial model or use default
32    let base_rate = calculate_base_rate(actuarial_model, coverage_type);
33
34    // Step 2: Calculate comprehensive risk multiplier
35    let risk_multiplier = calculate_risk_multiplier(risk_assessment);
36
37    // Step 3: Calculate coverage type multiplier
38    let coverage_multiplier = coverage_type_multiplier(coverage_type);
39
40    // Step 4: Calculate pool utilization adjustment
41    let pool_utilization_multiplier = calculate_pool_utilization_multiplier(pool);
42
43    // Step 5: Calculate time-based adjustments
44    let time_multiplier = calculate_time_multiplier(policy_duration_seconds);
45
46    // Step 6: Calculate discounts
47    let discount_multiplier = calculate_discount_multiplier(modifiers);
48
49    // Step 7: Rolling-window claim-frequency surcharge
50    // recent_claims_count = number of approved claims in the observation window
51    let claim_freq_multiplier =
52        calculate_claim_frequency_multiplier(modifiers.recent_claims_count);
53
54    // Formula: coverage × base_rate × risk × coverage × pool × time × discount × claim_freq
55    // All multipliers are in basis points (10 000 = 1.0×), divisor accounts for all 7 factors.
56    let annual_premium = coverage_amount
57        .saturating_mul(base_rate as u128)
58        .saturating_mul(risk_multiplier as u128)
59        .saturating_mul(coverage_multiplier as u128)
60        .saturating_mul(pool_utilization_multiplier as u128)
61        .saturating_mul(time_multiplier as u128)
62        .saturating_mul(discount_multiplier as u128)
63        .saturating_mul(claim_freq_multiplier as u128)
64        / PREMIUM_CALCULATION_DIVISOR;
65
66    // Prorate for policy duration
67    let duration_premium =
68        annual_premium.saturating_mul(policy_duration_seconds as u128) / SECONDS_PER_YEAR;
69
70    let monthly_premium = duration_premium / 12;
71
72    // Calculate dynamic deductible based on risk
73    let deductible = calculate_deductible(coverage_amount, risk_assessment, modifiers);
74
75    PremiumCalculation {
76        base_rate,
77        risk_multiplier,
78        coverage_multiplier,
79        pool_utilization_multiplier,
80        time_multiplier,
81        discount_multiplier,
82        claim_freq_multiplier,
83        annual_premium: duration_premium,
84        monthly_premium,
85        deductible,
86        breakdown: PremiumBreakdown {
87            base_premium: coverage_amount.saturating_mul(base_rate as u128)
88                / BASIS_POINTS_DENOMINATOR,
89            risk_adjustment: calculate_risk_adjustment_amount(
90                coverage_amount,
91                base_rate,
92                risk_multiplier,
93            ),
94            coverage_adjustment: calculate_coverage_adjustment_amount(
95                coverage_amount,
96                base_rate,
97                risk_multiplier,
98                coverage_multiplier,
99            ),
100            pool_adjustment: calculate_pool_adjustment_amount(
101                coverage_amount,
102                base_rate,
103                risk_multiplier,
104                coverage_multiplier,
105                pool_utilization_multiplier,
106            ),
107            time_adjustment: calculate_time_adjustment_amount(
108                coverage_amount,
109                base_rate,
110                risk_multiplier,
111                coverage_multiplier,
112                pool_utilization_multiplier,
113                time_multiplier,
114            ),
115            discount_amount: calculate_discount_amount(
116                coverage_amount,
117                base_rate,
118                risk_multiplier,
119                coverage_multiplier,
120                pool_utilization_multiplier,
121                time_multiplier,
122                discount_multiplier,
123            ),
124            claim_freq_adjustment: calculate_claim_freq_adjustment_amount(
125                coverage_amount,
126                base_rate,
127                risk_multiplier,
128                coverage_multiplier,
129                pool_utilization_multiplier,
130                time_multiplier,
131                discount_multiplier,
132                claim_freq_multiplier,
133            ),
134        },
135    }
136}

138/// Calculate base rate from actuarial model
139fn calculate_base_rate(
140    actuarial_model: Option<&ActuarialModel>,
141    coverage_type: &CoverageType,
142) -> u32 {
143    match actuarial_model {
144        Some(model) => {
145            // Use actuarial model: expected_loss_ratio * confidence_adjustment
146            // Expected loss ratio in basis points (e.g., 600 = 6%)
147            let expected_loss = model.expected_loss_ratio;
148
149            // Confidence level adjustment (95% = 1.0, 99% = 1.2)
150            let confidence_adjustment = match model.confidence_level {
151                95 => 100,
152                96 => 105,
153                97 => 110,
154                98 => 115,
155                99 => 120,
156                _ => 100,
157            };
158
159            // Base rate = expected_loss * confidence_adjustment / 100
160            let model_rate = expected_loss.saturating_mul(confidence_adjustment as u32) / 100;
161
162            // Add expense loading (20% for operational costs)
163            model_rate.saturating_mul(120) / 100
164        }
165        None => {
166            // Default rates by coverage type (in basis points)
167            coverage_type_base_rate(coverage_type)
168        }
169    }
170}

172/// Default base rates by coverage type
173fn coverage_type_base_rate(coverage_type: &CoverageType) -> u32 {
174    match coverage_type {
175        CoverageType::Fire => 120,            // 1.2%
176        CoverageType::Flood => 200,           // 2.0%
177        CoverageType::Earthquake => 250,      // 2.5%
178        CoverageType::Theft => 100,           // 1.0%
179        CoverageType::LiabilityDamage => 150, // 1.5%
180        CoverageType::NaturalDisaster => 220, // 2.2%
181        CoverageType::Comprehensive => 300,   // 3.0%
182    }
183}

185/// Calculate comprehensive risk multiplier from assessment scores
186fn calculate_risk_multiplier(assessment: &RiskAssessment) -> u32 {
187    // Weighted average of risk components
188    // Location risk: 30%, Construction risk: 25%, Age risk: 20%, Claims history: 25%
189    let weighted_score = assessment
190        .location_risk_score
191        .saturating_mul(30)
192        .saturating_add(assessment.construction_risk_score.saturating_mul(25))
193        .saturating_add(assessment.age_risk_score.saturating_mul(20))
194        .saturating_add(assessment.claims_history_score.saturating_mul(25))
195        / 100;
196
197    // Convert score (0-100) to multiplier (50-400 basis points)
198    // Score 0 = very high risk (4.0x), Score 100 = very low risk (0.5x)
199    match weighted_score {
200        0..=10 => 400,  // Very high risk
201        11..=20 => 350, // High risk
202        21..=30 => 300, // High-medium risk
203        31..=40 => 250, // Medium-high risk
204        41..=50 => 200, // Medium risk
205        51..=60 => 170, // Medium-low risk
206        61..=70 => 140, // Low-medium risk
207        71..=80 => 110, // Low risk
208        81..=90 => 85,  // Very low risk
209        _ => 60,        // Minimal risk
210    }
211}

226/// Pool utilization adjustment
227/// Higher utilization = higher premiums to manage risk
228fn calculate_pool_utilization_multiplier(pool: &RiskPool) -> u32 {
229    if pool.total_capital == 0 {
230        return 200; // Default high multiplier if no capital
231    }
232
233    // Utilization rate: (total_capital - available_capital) / total_capital
234    let utilized = pool.total_capital.saturating_sub(pool.available_capital);
235    let utilization_rate = utilized.saturating_mul(100) / pool.total_capital;
236
237    // Adjust multiplier based on utilization
238    match utilization_rate {
239        0..=30 => 90,   // Low utilization - discount
240        31..=50 => 100, // Normal utilization
241        51..=70 => 115, // Medium-high utilization - slight increase
242        71..=85 => 135, // High utilization - significant increase
243        _ => 160,       // Critical utilization - major increase
244    }
245}

247/// Time-based adjustment
248/// Longer policies get slight discounts for stability
249fn calculate_time_multiplier(duration_seconds: u64) -> u32 {
250    match duration_seconds {
251        0..=2_592_000 => 105,          // < 30 days - short term premium
252        2_592_001..=7_776_000 => 100,  // 1-3 months - standard
253        7_776_001..=15_552_000 => 95,  // 3-6 months - slight discount
254        15_552_001..=31_536_000 => 90, // 6-12 months - good discount
255        _ => 85,                       // > 1 year - best discount
256    }
257}

259/// Rolling-window claim-frequency surcharge multiplier.
260///
261/// Uses the number of approved claims filed within the rolling observation
262/// window (typically 12 months) stored in `PremiumModifiers::recent_claims_count`.
263///
264/// | Claims in window | Multiplier | Basis points |
265/// |-----------------|------------|-------------|
266/// | 0               | 1.00×      | 10 000       | (no surcharge — baseline)
267/// | 1               | 1.10×      | 11 000       |
268/// | 2               | 1.25×      | 12 500       |
269/// | 3               | 1.50×      | 15 000       |
270/// | 4               | 1.75×      | 17 500       |
271/// | 5+              | 2.00×      | 20 000       | (high-frequency cap)
272fn calculate_claim_frequency_multiplier(recent_claims_count: u32) -> u32 {
273    match recent_claims_count {
274        0 => 10_000,  // 1.00× — no surcharge
275        1 => 11_000,  // 1.10×
276        2 => 12_500,  // 1.25×
277        3 => 15_000,  // 1.50×
278        4 => 17_500,  // 1.75×
279        _ => 20_000,  // 2.00× — high-frequency cap (5+ claims)
280    }
281}

283/// Calculate discount multiplier from modifiers
284fn calculate_discount_multiplier(modifiers: &PremiumModifiers) -> u32 {
285    let mut total_discount_bps: u32 = 0;
286
287    // Multi-policy discount (up to 15%)
288    if modifiers.has_multiple_policies {
289        total_discount_bps = total_discount_bps.saturating_add(1500);
290    }
291
292    // Claim-free discount (up to 20% based on years)
293    if modifiers.claim_free_years > 0 {
294        let claim_free_discount = match modifiers.claim_free_years {
295            1 => 500,  // 5%
296            2 => 1000, // 10%
297            3 => 1500, // 15%
298            _ => 2000, // 20% for 4+ years
299        };
300        total_discount_bps = total_discount_bps.saturating_add(claim_free_discount);
301    }
302
303    // Safety features discount (up to 10%)
304    if modifiers.has_safety_features {
305        total_discount_bps = total_discount_bps.saturating_add(1000);
306    }
307
308    // Loyalty discount (up to 10%)
309    if modifiers.loyalty_years > 0 {
310        let loyalty_discount = match modifiers.loyalty_years {
311            1..=2 => 300, // 3%
312            3..=5 => 600, // 6%
313            _ => 1000,    // 10% for 6+ years
314        };
315        total_discount_bps = total_discount_bps.saturating_add(loyalty_discount);
316    }
317
318    // Cap total discount at 40%
319    if total_discount_bps > 4000 {
320        total_discount_bps = 4000;
321    }
322
323    // Convert discount to multiplier (10000 - discount_bps)
324    10_000u32.saturating_sub(total_discount_bps)
325}

327/// Calculate deductible based on risk and modifiers
328fn calculate_deductible(
329    coverage_amount: u128,
330    assessment: &RiskAssessment,
331    modifiers: &PremiumModifiers,
332) -> u128 {
333    // Base deductible: 5% of coverage
334    let base_deductible_rate: u32 = 500; // 5% in basis points
335
336    // Adjust based on risk (higher risk = higher deductible)
337    let risk_adjustment: u32 = match assessment.overall_risk_score {
338        0..=20 => 200,  // Very high risk - 20% deductible
339        21..=40 => 150, // High risk - 15%
340        41..=60 => 100, // Medium risk - 10%
341        61..=80 => 75,  // Low risk - 7.5%
342        _ => 50,        // Very low risk - 5%
343    };
344
345    let deductible_rate = base_deductible_rate.saturating_add(risk_adjustment);
346
347    // Apply safety feature reduction
348    let reduction: u32 = 50;
349    let final_rate = if modifiers.has_safety_features {
350        deductible_rate.saturating_sub(reduction)
351    } else {
352        deductible_rate
353    };
354
355    coverage_amount.saturating_mul(final_rate as u128) / 10_000
356}

358/// Calculate risk adjustment amount for breakdown
359fn calculate_risk_adjustment_amount(coverage: u128, base_rate: u32, risk_multiplier: u32) -> u128 {
360    let base_premium = coverage.saturating_mul(base_rate as u128) / BASIS_POINTS_DENOMINATOR;
361    let risk_adjusted =
362        base_premium.saturating_mul(risk_multiplier as u128) / BASIS_POINTS_DENOMINATOR;
363    risk_adjusted.saturating_sub(base_premium)
364}

366/// Calculate coverage adjustment amount (difference that coverage_multiplier adds)
367fn calculate_coverage_adjustment_amount(
368    coverage: u128,
369    base_rate: u32,
370    risk_multiplier: u32,
371    coverage_multiplier: u32,
372) -> u128 {
373    let premium_before = coverage
374        .saturating_mul(base_rate as u128)
375        .saturating_mul(risk_multiplier as u128)
376        / PREMIUM_CALCULATION_DIVISOR;
377
378    let premium_after =
379        premium_before.saturating_mul(coverage_multiplier as u128) / BASIS_POINTS_DENOMINATOR;
380
381    premium_after.saturating_sub(premium_before)
382}

426/// Calculate discount amount
427fn calculate_discount_amount(
428    coverage: u128,
429    base_rate: u32,
430    risk_multiplier: u32,
431    coverage_multiplier: u32,
432    pool_multiplier: u32,
433    time_multiplier: u32,
434    discount_multiplier: u32,
435) -> u128 {
436    let premium_before_discount = coverage
437        .saturating_mul(base_rate as u128)
438        .saturating_mul(risk_multiplier as u128)
439        .saturating_mul(coverage_multiplier as u128)
440        .saturating_mul(pool_multiplier as u128)
441        .saturating_mul(time_multiplier as u128)
442        / PREMIUM_CALCULATION_DIVISOR_5MULT;
443
444    let final_premium = premium_before_discount.saturating_mul(discount_multiplier as u128)
445        / BASIS_POINTS_DENOMINATOR;
446
447    premium_before_discount.saturating_sub(final_premium)
448}

450/// Calculate the claim-frequency surcharge amount for the breakdown.
451///
452/// This is the additional cost the claim-frequency multiplier adds on top of
453/// the already-discounted premium (i.e. it is always additive / non-negative).
454fn calculate_claim_freq_adjustment_amount(
455    coverage: u128,
456    base_rate: u32,
457    risk_multiplier: u32,
458    coverage_multiplier: u32,
459    pool_multiplier: u32,
460    time_multiplier: u32,
461    discount_multiplier: u32,
462    claim_freq_multiplier: u32,
463) -> u128 {
464    // Premium after discount, before claim-frequency step
465    let pre_freq = coverage
466        .saturating_mul(base_rate as u128)
467        .saturating_mul(risk_multiplier as u128)
468        .saturating_mul(coverage_multiplier as u128)
469        .saturating_mul(pool_multiplier as u128)
470        .saturating_mul(time_multiplier as u128)
471        .saturating_mul(discount_multiplier as u128)
472        / PREMIUM_CALCULATION_DIVISOR;
473
474    let post_freq = pre_freq
475        .saturating_mul(claim_freq_multiplier as u128)
476        / BASIS_POINTS_DENOMINATOR;
477
478    post_freq.saturating_sub(pre_freq)
479}

481// Constants
482const BASIS_POINTS_DENOMINATOR: u128 = 10_000;
483const SECONDS_PER_YEAR: u128 = 31_536_000; // 365 * 24 * 60 * 60
484/// Divisor for the 7-factor chain: base_rate × risk × coverage × pool × time × discount × claim_freq
485/// All 7 multipliers are in basis points (10 000 = 1.0×), so each divides by 10 000.
486/// 10 000^6 = 10^24; but base_rate uses a different scale (it's in raw basis points ÷ 10 000 once),
487/// so the combined divisor keeps the same structure as before with an extra ×10 000 for claim_freq.
488const PREMIUM_CALCULATION_DIVISOR: u128 = 10_000_000_000_000_000_000_000_000; // 10^25 for 7 factors
489const PREMIUM_CALCULATION_DIVISOR_LARGE: u128 = 10_000_000_000_000_000_000_000; // 10^22 for 6 multipliers (breakdown helpers)
490const PREMIUM_CALCULATION_DIVISOR_5MULT: u128 = 1_000_000_000_000_000_000_000; // 10^21 for discount calc (5 factors before discount)
