217        /// Creates a new Governance contract.
218        ///
219        /// # Arguments
220        /// * `signers` - Initial list of signer accounts
221        /// * `threshold` - Number of approvals required (must be <= signers.len())
222        /// * `timelock_blocks` - Blocks to wait after approval before execution
223        #[ink(constructor)]
224        pub fn new(signers: Vec<AccountId>, threshold: u32, timelock_blocks: u64) -> Self {
225            let caller = Self::env().caller();
226            let mut unique_signers = signers;
227            unique_signers.dedup();
228            let signer_count = unique_signers.len() as u32;
229            let safe_threshold = if threshold == 0 || threshold > signer_count {
230                signer_count
231            } else {
232                threshold
233            };
234
235            Self {
236                admin: caller,
237                signers: unique_signers,
238                threshold: safe_threshold,
239                proposal_counter: 0,
240                active_proposal_count: 0,
241                approved_proposal_count: 0,
242                executed_proposal_count: 0,
243                rejected_proposal_count: 0,
244                cancelled_proposal_count: 0,
245                closed_participation_count: 0,
246                participation_sum_bps: 0,
247                proposals: Mapping::default(),
248                votes: Mapping::default(),
249                timelock_blocks,
250                signer_public_keys: Mapping::default(),
251                pending_admin_rotation: None,
252                vote_commitments: Mapping::default(),
253                reveal_phase_started: Mapping::default(),
254                reveal_phase_duration: 10_800, // ~18 hours at 6s blocks
255                proposal_comments: Mapping::default(),
256            }
257        }

305        /// Creates a new governance proposal. Only signers may propose.
306        #[ink(message)]
307        pub fn create_proposal(
308            &mut self,
309            description_hash: Hash,
310            action_type: GovernanceAction,
311            target: Option<AccountId>,
312        ) -> Result<u64, Error> {
313            let caller = self.env().caller();
314            self.ensure_signer(caller)?;
315
316            if self.active_proposal_count >= constants::GOVERNANCE_MAX_ACTIVE_PROPOSALS {
317                return Err(Error::MaxProposals);
318            }
319
320            let proposal_id = self.proposal_counter;
321            self.proposal_counter = self.proposal_counter.saturating_add(1);
322            let now = self.env().block_number() as u64;
323
324            let proposal = GovernanceProposal {
325                id: proposal_id,
326                proposer: caller,
327                description_hash,
328                action_type: action_type.clone(),
329                target,
330                threshold: self.threshold,
331                votes_for: 0,
332                votes_against: 0,
333                status: ProposalStatus::Active,
334                created_at: now,
335                executed_at: 0,
336                timelock_until: 0,
337                is_emergency: false,
338            };
339
340            self.proposals.insert(proposal_id, &proposal);
341            self.active_proposal_count = self.active_proposal_count.saturating_add(1);
342
343            self.env().emit_event(ProposalCreated {
344                proposal_id,
345                proposer: caller,
346                action_type,
347                threshold: self.threshold,
348            });
349
350            Ok(proposal_id)
351        }

405        /// Returns the governance analytics.
406        ///
407        /// All figures are served from counters maintained incrementally at
408        /// each status transition (see `apply_status_transition`), so this is
409        /// O(1) storage reads regardless of how many proposals exist — the
410        /// previous implementation rescanned the entire proposal history on
411        /// every call, which grows without bound (Issue #972).
412        #[ink(message)]
413        pub fn get_analytics(&self) -> GovernanceAnalytics {
414            let avg_participation_bps = if self.closed_participation_count > 0 {
415                (self.participation_sum_bps / self.closed_participation_count) as u32
416            } else {
417                0
418            };
419
420            GovernanceAnalytics {
421                total_proposals: self.proposal_counter,
422                executed_proposals: self.executed_proposal_count,
423                rejected_proposals: self.rejected_proposal_count,
424                cancelled_proposals: self.cancelled_proposal_count,
425                active_proposals: self.active_proposal_count as u64 + self.approved_proposal_count,
426                avg_participation_bps,
427            }
428        }
429
430        /// Move `proposal` into `new_status`, keeping every analytics counter
431        /// consistent without rescanning proposal history (Issue #972).
432        ///
433        /// Handles arbitrary transitions uniformly (including emergency
434        /// overrides of already-rejected proposals): the counter for the old
435        /// status is decremented and the new status incremented. A proposal
436        /// contributes to the participation average exactly once — when it
437        /// first leaves the open set (`Active`/`Approved`) into a closing
438        /// status (`Executed`/`Rejected`), matching the historical behaviour
439        /// of the brute-force scan.
440        fn apply_status_transition(
441            &mut self,
442            proposal: &mut GovernanceProposal,
443            new_status: ProposalStatus,
444        ) {
445            // Decrement the bucket for the current status.
446            match proposal.status {
447                ProposalStatus::Active => {
448                    self.active_proposal_count = self.active_proposal_count.saturating_sub(1);
449                }
450                ProposalStatus::Approved => {
451                    self.approved_proposal_count = self.approved_proposal_count.saturating_sub(1);
452                }
453                ProposalStatus::Executed => {
454                    self.executed_proposal_count = self.executed_proposal_count.saturating_sub(1);
455                }
456                ProposalStatus::Rejected => {
457                    self.rejected_proposal_count = self.rejected_proposal_count.saturating_sub(1);
458                }
459                ProposalStatus::Cancelled => {
460                    self.cancelled_proposal_count = self.cancelled_proposal_count.saturating_sub(1);
461                }
462                ProposalStatus::Expired => {}
463            }
464
465            // Participation bookkeeping: count a closure only when leaving the
466            // open set, so re-opened/closed-again proposals are never double
467            // counted.
468            if !matches!(
469                proposal.status,
470                ProposalStatus::Executed | ProposalStatus::Rejected
471            ) && matches!(
472                new_status,
473                ProposalStatus::Executed | ProposalStatus::Rejected
474            ) {
475                let signer_count = self.signers.len() as u64;
476                if signer_count > 0 {
477                    let total_votes =
478                        (proposal.votes_for.saturating_add(proposal.votes_against)) as u64;
479                    let bps = total_votes.saturating_mul(10_000) / signer_count;
480                    self.participation_sum_bps = self.participation_sum_bps.saturating_add(bps);
481                }
482                self.closed_participation_count = self.closed_participation_count.saturating_add(1);
483            }
484
485            // Increment the bucket for the new status.
486            match new_status {
487                ProposalStatus::Active => {}
488                ProposalStatus::Approved => {
489                    self.approved_proposal_count = self.approved_proposal_count.saturating_add(1);
490                }
491                ProposalStatus::Executed => {
492                    self.executed_proposal_count = self.executed_proposal_count.saturating_add(1);
493                }
494                ProposalStatus::Rejected => {
495                    self.rejected_proposal_count = self.rejected_proposal_count.saturating_add(1);
496                }
497                ProposalStatus::Cancelled => {
498                    self.cancelled_proposal_count = self.cancelled_proposal_count.saturating_add(1);
499                }
500                ProposalStatus::Expired => {}
501            }
502
503            proposal.status = new_status;
504        }

512        /// Returns the participation rate for a specific proposal in basis points.
513        #[ink(message)]
514        pub fn get_proposal_participation(&self, proposal_id: u64) -> Result<u32, Error> {
515            let proposal = self
516                .proposals
517                .get(proposal_id)
518                .ok_or(Error::ProposalNotFound)?;
519            let signer_count = self.signers.len() as u32;
520            if signer_count == 0 {
521                return Ok(0);
522            }
523            let total_votes = proposal.votes_for.saturating_add(proposal.votes_against);
524            let bps = (total_votes as u64).saturating_mul(10_000) / (signer_count as u64);
525            Ok(bps as u32)
526        }

528        /// Casts a vote on an active proposal. Only signers may vote.
529        #[ink(message)]
530        pub fn vote(&mut self, proposal_id: u64, support: bool) -> Result<(), Error> {
531            let caller = self.env().caller();
532            self.ensure_signer(caller)?;
533
534            let mut proposal = self
535                .proposals
536                .get(proposal_id)
537                .ok_or(Error::ProposalNotFound)?;
538
539            if proposal.status != ProposalStatus::Active {
540                return Err(Error::ProposalClosed);
541            }
542
543            if self.votes.contains((proposal_id, caller)) {
544                return Err(Error::AlreadyVoted);
545            }
546
547            self.votes.insert((proposal_id, caller), &support);
548            if support {
549                proposal.votes_for = proposal.votes_for.saturating_add(1);
550            } else {
551                proposal.votes_against = proposal.votes_against.saturating_add(1);
552            }
553
554            // Check if threshold reached → move to Approved with timelock
555            if proposal.votes_for >= proposal.threshold {
556                let now = self.env().block_number() as u64;
557                let timelock = if proposal.is_emergency {
558                    now // Bypass timelock
559                } else {
560                    now.saturating_add(self.timelock_blocks)
561                };
562                proposal.timelock_until = timelock;
563                self.apply_status_transition(&mut proposal, ProposalStatus::Approved);
564            }
565
566            // Check if rejection is certain (remaining votes can't reach threshold)
567            let total_signers = self.signers.len() as u32;
568            let total_votes = proposal.votes_for.saturating_add(proposal.votes_against);
569            let remaining = total_signers.saturating_sub(total_votes);
570            if proposal.votes_for.saturating_add(remaining) < proposal.threshold {
571                self.apply_status_transition(&mut proposal, ProposalStatus::Rejected);
572                self.env().emit_event(ProposalRejected { proposal_id });
573            }
574
575            self.proposals.insert(proposal_id, &proposal);
576
577            self.env().emit_event(VoteCast {
578                proposal_id,
579                voter: caller,
580                support,
581            });
582
583            Ok(())
584        }

603        /// Vote with optional ECDSA cryptographic signature verification.
604        #[ink(message)]
605        pub fn vote_with_signature(
606            &mut self,
607            proposal_id: u64,
608            support: bool,
609            signed_approval: Option<propchain_traits::SignedApproval>,
610        ) -> Result<(), Error> {
611            let caller = self.env().caller();
612
613            if let Some(ref approval) = signed_approval {
614                let expected_key = self
615                    .signer_public_keys
616                    .get(caller)
617                    .ok_or(Error::Unauthorized)?;
618                propchain_traits::crypto::verify_signed_approval(approval, &expected_key)
619                    .map_err(|_| Error::Unauthorized)?;
620
621                let expected_hash = propchain_traits::crypto::hash_encoded(&(
622                    proposal_id,
623                    support,
624                    caller,
625                    self.env().block_number(),
626                ));
627                if approval.message_hash != <[u8; 32]>::from(expected_hash) {
628                    return Err(Error::Unauthorized);
629                }
630            }
631
632            self.vote(proposal_id, support)
633        }

635        /// Executes an approved proposal after the timelock has elapsed.
636        #[ink(message)]
637        pub fn execute_proposal(&mut self, proposal_id: u64) -> Result<(), Error> {
638            let caller = self.env().caller();
639            self.ensure_signer(caller)?;
640
641            let mut proposal = self
642                .proposals
643                .get(proposal_id)
644                .ok_or(Error::ProposalNotFound)?;
645
646            if proposal.status != ProposalStatus::Approved {
647                return Err(Error::ProposalClosed);
648            }
649
650            let now = self.env().block_number() as u64;
651            if now < proposal.timelock_until {
652                return Err(Error::TimelockActive);
653            }
654
655            self.apply_status_transition(&mut proposal, ProposalStatus::Executed);
656            proposal.executed_at = now;
657            self.proposals.insert(proposal_id, &proposal);
658
659            self.env().emit_event(ProposalExecuted {
660                proposal_id,
661                executed_at: now,
662            });
663
664            Ok(())
665        }

669        /// Submit a hashed commitment for a private vote.
670        /// The commitment should be hash(proposal_id || voter || support || salt).
671        #[ink(message)]
672        pub fn commit_vote(&mut self, proposal_id: u64, commitment: Hash) -> Result<(), Error> {
673            let caller = self.env().caller();
674            self.ensure_signer(caller)?;
675
676            let proposal = self
677                .proposals
678                .get(proposal_id)
679                .ok_or(Error::ProposalNotFound)?;
680
681            if proposal.status != ProposalStatus::Active {
682                return Err(Error::ProposalClosed);
683            }
684
685            if self.vote_commitments.contains((proposal_id, caller)) {
686                return Err(Error::AlreadyVoted);
687            }
688
689            self.vote_commitments
690                .insert((proposal_id, caller), &commitment);
691
692            Ok(())

695        /// Start the reveal phase for a proposal (any signer may call).
696        #[ink(message)]
697        pub fn start_reveal_phase(&mut self, proposal_id: u64) -> Result<(), Error> {
698            let caller = self.env().caller();
699            self.ensure_signer(caller)?;
700
701            let proposal = self
702                .proposals
703                .get(proposal_id)
704                .ok_or(Error::ProposalNotFound)?;
705
706            if proposal.status != ProposalStatus::Active {
707                return Err(Error::ProposalClosed);
708            }
709
710            if self.reveal_phase_started.get(proposal_id).unwrap_or(false) {
711                return Err(Error::AlreadyVoted);
712            }
713
714            self.reveal_phase_started.insert(proposal_id, &true);
715            Ok(())
716        }

718        /// Reveal a private vote after the commitment phase.
719        /// Verifies that the revealed vote matches the earlier commitment.
720        #[ink(message)]
721        pub fn reveal_vote(
722            &mut self,
723            proposal_id: u64,
724            support: bool,
725            salt: [u8; 32],
726        ) -> Result<(), Error> {
727            let caller = self.env().caller();
728            self.ensure_signer(caller)?;
729
730            if !self.reveal_phase_started.get(proposal_id).unwrap_or(false) {
731                return Err(Error::ProposalClosed);
732            }
733
734            let commitment = self
735                .vote_commitments
736                .get((proposal_id, caller))
737                .ok_or(Error::AlreadyVoted)?;
738
739            // Verify the commitment matches
740            let encoded = (proposal_id, caller, support, salt);
741            let expected = propchain_traits::crypto::hash_encoded(&encoded);
742            if commitment != expected {
743                return Err(Error::Unauthorized);
744            }
745
746            // Clear commitment to prevent double-reveal
747            self.vote_commitments.remove((proposal_id, caller));
748
749            // Record the vote via internal logic
750            self.record_vote(proposal_id, caller, support)?;
751
752            Ok(())
753        }

755        /// Cancels an active proposal. Only the proposer or admin may cancel.
756        #[ink(message)]
757        pub fn cancel_proposal(&mut self, proposal_id: u64) -> Result<(), Error> {
758            let caller = self.env().caller();
759            let mut proposal = self
760                .proposals
761                .get(proposal_id)
762                .ok_or(Error::ProposalNotFound)?;
763
764            if proposal.status != ProposalStatus::Active
765                && proposal.status != ProposalStatus::Approved
766            {
767                return Err(Error::ProposalClosed);
768            }
769
770            if caller != proposal.proposer && caller != self.admin {
771                return Err(Error::Unauthorized);
772            }
773
774            self.apply_status_transition(&mut proposal, ProposalStatus::Cancelled);
775            self.proposals.insert(proposal_id, &proposal);
776
777            Ok(())
778        }

780        /// Adds a new signer. Only admin may call.
781        ///
782        /// # Roster freeze policy (Issue #1024)
783        ///
784        /// The signer roster is frozen while any proposal is `Active`. `vote`
785        /// and `get_proposal_participation` read the **live** signer count, so
786        /// changing the roster mid-vote would silently alter a proposal's
787        /// quorum math (a removed signer can shrink `total_signers` and force
788        /// an early rejection; an added signer can keep a doomed proposal
789        /// alive). Admin roster changes are therefore rejected until every
790        /// active proposal has closed; use `emergency_override` for urgent
791        /// cases.
792        #[ink(message)]
793        pub fn add_signer(&mut self, new_signer: AccountId) -> Result<(), Error> {
794            self.ensure_admin()?;
795
796            if self.active_proposal_count > 0 {
797                return Err(Error::SignerChangesLocked);
798            }
799
800            if self.signers.contains(&new_signer) {
801                return Err(Error::SignerExists);
802            }
803
804            if self.signers.len() as u32 >= constants::GOVERNANCE_MAX_SIGNERS {
805                return Err(Error::MaxProposals);
806            }
807
808            self.signers.push(new_signer);
809
810            self.env().emit_event(SignerAdded {
811                signer: new_signer,
812                added_by: self.env().caller(),
813            });
814
815            Ok(())
816        }
817
818        /// Removes a signer. Only admin may call.
819        ///
820        /// # Roster freeze policy (Issue #1024)
821        ///
822        /// Same freeze as `add_signer`: the roster cannot change while any
823        /// proposal is `Active`, so a proposal's pass/reject outcome never
824        /// depends on when the admin happened to edit the signer set.
825        #[ink(message)]
826        pub fn remove_signer(&mut self, signer: AccountId) -> Result<(), Error> {
827            self.ensure_admin()?;
828
829            if self.active_proposal_count > 0 {
830                return Err(Error::SignerChangesLocked);
831            }
832
833            if self.signers.len() as u32 <= constants::GOVERNANCE_MIN_SIGNERS {
834                return Err(Error::MinSigners);
835            }
836
837            let pos = self
838                .signers
839                .iter()
840                .position(|s| *s == signer)
841                .ok_or(Error::SignerNotFound)?;
842
843            self.signers.swap_remove(pos);
844
845            // Adjust threshold if it's now greater than signer count
846            let new_count = self.signers.len() as u32;
847            if self.threshold > new_count {
848                let old = self.threshold;
849                self.threshold = new_count;
850                self.env().emit_event(ThresholdUpdated {
851                    old_threshold: old,
852                    new_threshold: new_count,
853                });
854            }
855
856            self.env().emit_event(SignerRemoved {
857                signer,
858                removed_by: self.env().caller(),
859            });
860
861            Ok(())
862        }

865        #[ink(message)]
866        pub fn update_threshold(&mut self, new_threshold: u32) -> Result<(), Error> {
867            self.ensure_admin()?;

869            if new_threshold == 0 || new_threshold > self.signers.len() as u32 {
870                return Err(Error::InvalidThreshold);
871            }
872
873            let old = self.threshold;
874            self.threshold = new_threshold;

877                old_threshold: old,
878                new_threshold,
879            });

881            Ok(())
882        }

884        /// Emergency override: admin can force-execute or reject a proposal.
885        #[ink(message)]
886        pub fn emergency_override(&mut self, proposal_id: u64, execute: bool) -> Result<(), Error> {
887            self.ensure_admin()?;
888
889            let mut proposal = self
890                .proposals
891                .get(proposal_id)
892                .ok_or(Error::ProposalNotFound)?;
893
894            if proposal.status == ProposalStatus::Executed
895                || proposal.status == ProposalStatus::Cancelled
896            {
897                return Err(Error::ProposalClosed);
898            }
899
900            let now = self.env().block_number() as u64;
901            if execute {
902                self.apply_status_transition(&mut proposal, ProposalStatus::Executed);
903                proposal.executed_at = now;
904            } else {
905                self.apply_status_transition(&mut proposal, ProposalStatus::Rejected);
906            }
907
908            self.proposals.insert(proposal_id, &proposal);
909
910            self.env().emit_event(EmergencyOverrideUsed {
911                proposal_id,
912                admin: self.env().caller(),
913            });
914
915            Ok(())
916        }

918        /// Request a two-step admin rotation with cooldown.
919        #[ink(message)]
920        pub fn request_admin_rotation(&mut self, new_admin: AccountId) -> Result<(), Error> {
921            self.ensure_admin()?;
922            let caller = self.env().caller();
923            let block = self.env().block_number();
924            let effective_at =
925                block.saturating_add(propchain_traits::constants::KEY_ROTATION_COOLDOWN_BLOCKS);
926
927            self.pending_admin_rotation = Some(propchain_traits::KeyRotationRequest {
928                old_account: caller,
929                new_account: new_admin,
930                requested_at: block,
931                effective_at,
932                confirmed: false,
933            });
934
935            Ok(())
936        }

938        /// Confirm a pending admin rotation after cooldown.
939        #[ink(message)]
940        pub fn confirm_admin_rotation(&mut self) -> Result<(), Error> {
941            let caller = self.env().caller();
942            let block = self.env().block_number();
943
944            let request = self
945                .pending_admin_rotation
946                .as_ref()
947                .ok_or(Error::ProposalNotFound)?;
948
949            if request.new_account != caller {
950                return Err(Error::Unauthorized);
951            }
952            if block < request.effective_at {
953                return Err(Error::TimelockActive);
954            }
955            let expiry = request
956                .effective_at
957                .saturating_add(propchain_traits::constants::KEY_ROTATION_EXPIRY_BLOCKS);
958            if block > expiry {
959                self.pending_admin_rotation = None;
960                return Err(Error::ProposalExpired);
961            }
962
963            self.admin = caller;
964            self.pending_admin_rotation = None;
965            Ok(())
966        }

969        #[ink(message)]
970        pub fn cancel_admin_rotation(&mut self) -> Result<(), Error> {
971            let caller = self.env().caller();
972            let request = self
973                .pending_admin_rotation
974                .as_ref()
975                .ok_or(Error::ProposalNotFound)?;
976
977            if caller != request.old_account && caller != request.new_account {

981            self.pending_admin_rotation = None;

1001        /// Internal vote recording logic shared by `vote` and `reveal_vote`.
1002        fn record_vote(
1003            &mut self,
1004            proposal_id: u64,
1005            caller: AccountId,
1006            support: bool,
1007        ) -> Result<(), Error> {
1008            let mut proposal = self
1009                .proposals
1010                .get(proposal_id)
1011                .ok_or(Error::ProposalNotFound)?;
1012
1013            if proposal.status != ProposalStatus::Active {
1014                return Err(Error::ProposalClosed);
1015            }
1016
1017            if self.votes.contains((proposal_id, caller)) {
1018                return Err(Error::AlreadyVoted);
1019            }
1020
1021            self.votes.insert((proposal_id, caller), &support);
1022            if support {
1023                proposal.votes_for = proposal.votes_for.saturating_add(1);
1024            } else {
1025                proposal.votes_against = proposal.votes_against.saturating_add(1);
1026            }
1027
1028            // Check if threshold reached → move to Approved with timelock
1029            if proposal.votes_for >= proposal.threshold {
1030                let now = self.env().block_number() as u64;
1031                let timelock = if proposal.is_emergency {
1032                    now
1033                } else {
1034                    now.saturating_add(self.timelock_blocks)
1035                };
1036                proposal.timelock_until = timelock;
1037                self.apply_status_transition(&mut proposal, ProposalStatus::Approved);
1038            }
1039
1040            // Check if rejection is certain
1041            let total_signers = self.signers.len() as u32;
1042            let total_votes = proposal.votes_for.saturating_add(proposal.votes_against);
1043            let remaining = total_signers.saturating_sub(total_votes);
1044            if proposal.votes_for.saturating_add(remaining) < proposal.threshold {
1045                self.apply_status_transition(&mut proposal, ProposalStatus::Rejected);
1046                self.env().emit_event(ProposalRejected { proposal_id });
1047            }
1048
1049            self.proposals.insert(proposal_id, &proposal);
1050            Ok(())
1051        }
